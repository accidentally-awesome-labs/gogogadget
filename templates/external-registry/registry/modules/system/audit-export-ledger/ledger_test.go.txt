// The contract table an external adapter owes the slot it implements. It
// installs with the module, so a consumer who adds this adapter also gets
// the proof that it satisfies audit.Exporter — the same table the core
// adapters are held to, run against a third party's code.

func TestLedgerSatisfiesTheExporterContract(t *testing.T) {
	var exporter audit.Exporter = NewFileExporter(t.TempDir())
	require.NotNil(t, exporter)
}

func TestFileTargetAppendsOneRecordPerEntry(t *testing.T) {
	dir := t.TempDir()
	exporter := NewFileExporter(dir)
	t.Cleanup(func() { _ = exporter.Close() })

	assert.Equal(t, meta.TargetFile, exporter.Target())
	for _, action := range []string{"project.created", "project.deleted"} {
		require.NoError(t, exporter.Export(context.Background(), audit.Entry{
			ID: "aud_" + action, OrgID: "org_1", UserID: "usr_1", Action: action,
			Metadata: map[string]any{"source": "contract-test"},
		}))
	}

	raw, err := os.ReadFile(filepath.Join(dir, ledgerFile))
	require.NoError(t, err)
	lines := strings.Split(strings.TrimSpace(string(raw)), "\n")
	require.Len(t, lines, 2)

	var first record
	require.NoError(t, json.Unmarshal([]byte(lines[0]), &first))
	assert.Equal(t, "project.created", first.Action)
	assert.Equal(t, "org_1", first.OrgID)
	assert.Equal(t, "usr_1", first.UserID)
	assert.Equal(t, "contract-test", first.Metadata["source"])
}

func TestCloudTargetPostsWithBearerToken(t *testing.T) {
	var seen []byte
	var authorization string
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		authorization = r.Header.Get("Authorization")
		seen, _ = io.ReadAll(r.Body)
		w.WriteHeader(http.StatusAccepted)
	}))
	t.Cleanup(server.Close)

	exporter := NewCloudExporter(server.URL, "tok_live")
	assert.Equal(t, meta.TargetCloud, exporter.Target())
	require.NoError(t, exporter.Export(context.Background(), audit.Entry{ID: "aud_1", Action: "org.updated"}))

	assert.Equal(t, "Bearer tok_live", authorization)
	var posted record
	require.NoError(t, json.Unmarshal(seen, &posted))
	assert.Equal(t, "org.updated", posted.Action)
}

// A managed target with no credential is an error the caller sees, never a
// silent local fallback: the audit ledger a compliance reviewer reads must
// not quietly become a file in tmp/.
func TestCloudTargetWithoutTokenRefuses(t *testing.T) {
	err := (&Exporter{Endpoint: "https://ledger.gadgetworks.example/v1/events"}).
		Export(context.Background(), audit.Entry{Action: "x"})
	require.Error(t, err)
	assert.Contains(t, err.Error(), meta.EnvToken)
}

func TestProviderErrorsPropagateToTheCaller(t *testing.T) {
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
		w.WriteHeader(http.StatusInternalServerError)
	}))
	t.Cleanup(server.Close)

	err := NewCloudExporter(server.URL, "tok_live").Export(context.Background(), audit.Entry{Action: "x"})
	require.Error(t, err)
	assert.Contains(t, err.Error(), "500")
}

func TestExportHonorsACancelledContext(t *testing.T) {

	cancel()
	err := NewFileExporter(t.TempDir()).Export(ctx, audit.Entry{Action: "x"})
	require.ErrorIs(t, err, context.Canceled)
}

// Half of the lifecycle contract every buffering adapter owes: the close runs
// at most once and a second caller gets the same answer, because shutdown
// paths reach Stop more than once.
func TestModuleStopIsIdempotent(t *testing.T) {
	module, err := NewModule(context.Background(), nil, Deps{Config: &config.Config{
		Env: "development", GadgetworksLedgerDir: t.TempDir(),
	}})
	require.NoError(t, err)
	require.Equal(t, meta.TargetFile, module.Value.Target())
	require.NoError(t, module.Value.Export(context.Background(), audit.Entry{Action: "x"}))

	require.NoError(t, module.Stop(context.Background()))
	require.NoError(t, module.Stop(context.Background()))

	err = module.Value.Export(context.Background(), audit.Entry{Action: "after-close"})
	require.Error(t, err)
	assert.Contains(t, err.Error(), "closed")
}

// The other half: a close that has not finished must not outlive the
// caller's deadline. The once is consumed without ever closing the channel,
// which is exactly the shape of a shutdown blocked on a stuck provider.
func TestModuleStopHonorsTheCallersDeadline(t *testing.T) {
	blocked := &Module{Value: NewFileExporter(t.TempDir()), closed: make(chan struct{})}
	blocked.closeOnce.Do(func() {})

	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Millisecond)
	t.Cleanup(cancel)
	require.ErrorIs(t, blocked.Stop(ctx), context.DeadlineExceeded)
}

// The selected target comes from the project's provider choice, expressed as
// the environment the manifest allows each target in. It is never inferred
// from which credentials happen to be set.
func TestSelectedTargetFollowsTheEnvironmentNotTheCredentials(t *testing.T) {
	assert.Equal(t, meta.TargetFile, SelectedTarget("development"))
	assert.Equal(t, meta.TargetFile, SelectedTarget("test"))
	assert.Equal(t, meta.TargetCloud, SelectedTarget("production"))

	// A development run with a full set of cloud credentials still runs the
	// file target: the selection is not a credential sniff.
	module, err := NewModule(context.Background(), nil, Deps{Config: &config.Config{
		Env:                       "development",
		GadgetworksLedgerDir:      t.TempDir(),
		GadgetworksLedgerEndpoint: "https://ledger.gadgetworks.example",
		GadgetworksLedgerToken:    "tok_live",
	}})
	require.NoError(t, err)
	assert.Equal(t, meta.TargetFile, module.Value.Target())
}

// The managed selection is refused at boot, not at the first export, and
// never degrades to the local file: an operator running production who forgot
// the endpoint or the token is told during startup, with the key named.
func TestNewModuleRefusesAnIncompleteManagedSelection(t *testing.T) {
	missingEndpoint, err := NewModule(context.Background(), nil, Deps{Config: &config.Config{
		Env: "production", GadgetworksLedgerToken: "tok_live", GadgetworksLedgerDir: t.TempDir(),
	}})
	require.Error(t, err)
	require.Nil(t, missingEndpoint)
	assert.Contains(t, err.Error(), meta.EnvEndpoint)
	assert.Contains(t, err.Error(), meta.TargetCloud)

	missingToken, err := NewModule(context.Background(), nil, Deps{Config: &config.Config{
		Env: "production", GadgetworksLedgerEndpoint: "https://ledger.gadgetworks.example",
	}})
	require.Error(t, err)
	require.Nil(t, missingToken)
	assert.Contains(t, err.Error(), meta.EnvToken)
}

// The declared targets and slot are the manifest's, and code that branches
// on a target name must use the declared spelling.
func TestDeclaredIdentityIsStable(t *testing.T) {
	assert.Equal(t, "gadgetworks/system/audit-export-ledger", meta.ModuleID)
	assert.Equal(t, "ggg/audit-export", meta.Slot)
	assert.Equal(t, "ledger-file", meta.TargetFile)
	assert.Equal(t, "ledger-cloud", meta.TargetCloud)
	assert.Equal(t, "GADGETWORKS_LEDGER_DIR", meta.EnvDir)
	assert.Equal(t, "GADGETWORKS_LEDGER_ENDPOINT", meta.EnvEndpoint)
	assert.Equal(t, "GADGETWORKS_LEDGER_TOKEN", meta.EnvToken)
}
